#include<iostream>
#include<bits/stdc++.h>

using namespace std;

bool validMountainArray(vector<int>& arr) {

    int n = arr.size();
    
    if(n < 3) {

        return false;

    }

    for(int i = 0; i < n - 1; i++) {

        if(arr[(n / 2)] == 0) {

            return false;

        }

        if (arr[i + 1] > arr[i]) {

            continue;

        }

        else if (arr[i + 1] == arr[i]) {

            return false;

        } 

        else {

            continue;

        }

    }

    return true;
}

int main () {

    int n;

    cout << "Enter size of array: ";
    cin >> n;

    vector<int> arr(n);

    cout << "Enter array elements: " << endl;

    for(int i = 0; i < n; i++) {

        cin >> arr[i];

    }

    cout << validMountainArray (arr);
    
    return 0;

}
