            max = nums[i];

            if(i > 1 && nums[i - 1] == nums[i - 2]) {

                thirdmax = nums[i - 3];

            }

            else if(i > 1 && nums[i - 1] != nums[i - 2]) {

                thirdmax = nums[i - 2];

            }
            
            else {
                
                thirdmax = nums[i];
            
            }

        }

    }
};
