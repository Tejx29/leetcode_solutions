class Solution {
public:
    vector<int> findDisappearedNumbers(vector<int>& nums) {
        
        int n = nums.size();
    
        set<int> s1;
        vector<int> missing;

        for(int i = 0; i < n; i++) {

            s1.emplace(nums[i]);

        }

        // for(int i = 1; i <= n; i++) {

        //     cout << mp[i] << " ";

        // }

        for(int i = 1; i <= n; i++) {

            if(s1.count(i)) {

                continue;

            }

            missing.push_back(i);
        }

        return {missing.begin(), missing.end()};

    }
};
