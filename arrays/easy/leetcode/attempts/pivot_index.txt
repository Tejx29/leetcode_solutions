int pivot_index (vector<int> &arr) {

    int n = arr.size();
    int i = 0;
    int j = n - 1;

    int lsum = 0;
    int rsum = 0;

    while(j > i) {
        
        cout << "i: " << i << " j: " << j << endl;
   
        cout << "lsum: " << lsum << " rsum: " << rsum << endl;

        cout << endl;

        if(lsum == rsum && lsum != 0 && rsum != 0) {

            return i + 1;

        }
        
        if(arr[i] < arr[j] ) {
            
            lsum += arr[i];
            i++;
            continue;
            
        }

        else if(arr[i] > arr[j]) {

            rsum += arr[j];
            j--;
            continue;
        }

        lsum += arr[i];
        rsum += arr[j];
        i++;
        j--;

    }

    return -1;
}

class Solution {
public:
    int pivotIndex(vector<int>& nums) {

        int n = nums.size();
    
        int total = 0;

        int lsum = 0;
        int rsum = 0;

        for(auto it : nums) {

            total += it;

        }

        for(int i = 0; i < n; i++) {

            rsum = total - lsum - nums[i];

            if(lsum == rsum) {

                return i;

            }

            lsum += nums[i];
        }

        return -1;

    }
};
