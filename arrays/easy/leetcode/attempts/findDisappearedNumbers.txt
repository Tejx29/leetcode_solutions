class Solution {
public:
    vector<int> findDisappearedNumbers(vector<int>& nums) {
        
        int n = nums.size();
    
        set<int> s1;
        vector<int> missing;

        for(int i = 0; i < n; i++) {

            s1.emplace(nums[i]);

        }

        for(int i = 1; i <= n; i++) {

            if(s1.count(i)) {

                continue;

            }

            missing.push_back(i);
        }

        return {missing.begin(), missing.end()};

    }
};

class Solution {

    vector<int> findDisappearedNumbers(vector<int>& nums) {
        
        int n = nums.size();

        vector<int> missing;
        
        int visited;

        for(int i = 0; i < n; i++) {

            if(nums[i] > 0) {
                
                visited = (nums[i]) - 1;

                if(nums[visited] >= 0) {

                    nums[visited] = -nums[visited];

                }
            
            }

            else {
                
                nums[i] = -(nums[i]);

                visited = (nums[i]) - 1;

                if(nums[visited] >= 0) {

                    nums[visited] = -nums[visited];

                }

                if(nums[i] > 0) {
                
                    nums[i] = -(nums[i]);
            
                }

            }

        }

        for(int i = 0; i < n; i++) {

            if(nums[i] > 0) {

                missing.push_back(i + 1);

        }
        
        return missing;
    
    }
};

class Solution {
public:
    vector<int> findDisappearedNumbers(vector<int>& nums) {

        for(int i = 0; i < nums.size(); i++) {

            int index = abs(nums[i]) - 1;

            if(nums[index] > 0) {
                nums[index] = -nums[index];
            }
        }

        vector<int> ans;

        for(int i = 0; i < nums.size(); i++) {

            if(nums[i] > 0) {
                ans.push_back(i + 1);
            }
        }

        return ans;
    }
};
