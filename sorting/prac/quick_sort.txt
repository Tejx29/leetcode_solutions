void quick_sort(vector <int> &arr, int n, int low, int high) {

    int pivot = arr[low];

    int i = low, j = high;

    int count = 0;

    for(int i = low; i < n; i++) {

        cout << "* " << i << endl;

        while(arr[i] > pivot) {

            for(j = high; j > 0; j--) {

                if(j < i && arr[j] < pivot) {

                    swap(arr[j], arr[low]);

                    return;

                }

                if(arr[j] < pivot) {

                    swap(arr[j], arr[i]);

                    cout << endl;

                    break;
                }
            }
        
            break;
        }
    }
}
